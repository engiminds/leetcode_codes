            // Correct partition
            if (left1 <= right2 && left2 <= right1) {

                // Odd total length
                if ((m + n) % 2 == 1) {
                    return max(left1, left2);
                }

                // Even total length
                return (max(left1, left2) + min(right1, right2)) / 2.0;
            }

            // nums1 partition is too far right
            if (left1 > right2) {
                right = partition1 - 1;
            }
            // nums1 partition is too far left
            else {
                left = partition1 + 1;
            }
        }

        return 0.0;
    }
};
